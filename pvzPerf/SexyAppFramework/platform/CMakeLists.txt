cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(SexyFrameworkNativeWindow LANGUAGES C CXX)
    include(CTest)
    set(PVZ_WINDOW_BACKEND "X11" CACHE STRING "Native Linux window backend")
    set_property(CACHE PVZ_WINDOW_BACKEND PROPERTY STRINGS X11 WAYLAND)
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "This independent native window target currently implements Linux only")
endif()
find_package(PkgConfig REQUIRED)
if(PVZ_WINDOW_BACKEND STREQUAL "X11")
    find_package(X11 REQUIRED)
    add_library(sexy_native_window STATIC linux/X11Window.cpp)
    target_link_libraries(sexy_native_window PRIVATE X11::X11)
elseif(PVZ_WINDOW_BACKEND STREQUAL "WAYLAND")
    pkg_check_modules(WAYLAND REQUIRED IMPORTED_TARGET wayland-client)
    pkg_check_modules(WAYLAND_CURSOR REQUIRED IMPORTED_TARGET wayland-cursor)
    pkg_check_modules(XKBCOMMON REQUIRED IMPORTED_TARGET xkbcommon)
    pkg_check_modules(LIBDECOR REQUIRED IMPORTED_TARGET libdecor-0)
    add_library(sexy_native_window STATIC linux/WaylandWindow.cpp)
    target_link_libraries(sexy_native_window PRIVATE
        PkgConfig::WAYLAND PkgConfig::WAYLAND_CURSOR PkgConfig::XKBCOMMON PkgConfig::LIBDECOR)
else()
    message(FATAL_ERROR "PVZ_WINDOW_BACKEND must be X11 or WAYLAND for this target")
endif()
target_sources(sexy_native_window PRIVATE linux/ErrorDialog.cpp)
target_compile_definitions(sexy_native_window PRIVATE
    "$<$<BOOL:${PVZ_ENABLE_GAME_PROBES}>:PVZ_ENABLE_GAME_PROBES=1>")
target_compile_features(sexy_native_window PUBLIC cxx_std_20)
target_include_directories(sexy_native_window PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(sexy_native_window PRIVATE
    $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic;-Werror>)
option(PVZ_NATIVE_WINDOW_SANITIZERS "Enable ASan/UBSan for independent window gates" OFF)
if(PVZ_NATIVE_WINDOW_SANITIZERS)
    target_compile_options(sexy_native_window PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(sexy_native_window PUBLIC -fsanitize=address,undefined)
endif()
if(BUILD_TESTING)
    add_executable(sexy_native_window_tests tests/window_tests.cpp)
    target_link_libraries(sexy_native_window_tests PRIVATE sexy_native_window)
    if(PVZ_WINDOW_BACKEND STREQUAL "X11")
        target_compile_definitions(sexy_native_window_tests PRIVATE PVZ_WINDOW_TEST_X11=1)
        target_link_libraries(sexy_native_window_tests PRIVATE X11::X11)
    else()
        target_compile_definitions(sexy_native_window_tests PRIVATE PVZ_WINDOW_TEST_WAYLAND=1)
        target_link_libraries(sexy_native_window_tests PRIVATE PkgConfig::WAYLAND)
    endif()
    add_executable(sexy_pointer_snapshot_tests tests/pointer_snapshot_tests.cpp)
    target_link_libraries(sexy_pointer_snapshot_tests PRIVATE sexy_native_window)
    add_test(NAME sexy_pointer_snapshot COMMAND sexy_pointer_snapshot_tests)
    add_test(NAME sexy_native_window COMMAND sexy_native_window_tests)
    set_tests_properties(sexy_native_window PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
endif()
